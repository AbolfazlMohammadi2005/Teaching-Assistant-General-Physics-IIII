	\section{سوال اول}
	
	طول موج دوبروي ذرات زير را محاسبه کنيد:
	
	\begin{enumerate}
		\item[(الف)] يک پروتون با انرژي \(7\,\mathrm{MeV}\)
		\item[(ب)] يک الکترون با انرژي \(45\,\mathrm{GeV}\)
		\item[(ج)] يک الکترون با سرعت \(1.35 \times 10^6\,\mathrm{m/s}\)
	\end{enumerate}
	
	\section{سوال دوم}
	
	نوترون هاي حرارتي در يک راکتور هسته اي که به صورت
	\[
	K = 32 kT
	\]
	در نظر گرفته مي شوند (با \(T = 293\,\mathrm{K}\))، داراي ويژگي هاي زير هستند:
	
	\begin{enumerate}
		\item[(الف)] انرژي جنبشي اين نوترون ها را محاسبه کنيد.
		\item[(ب)] طول موج دوبروي آن ها را به دست آوريد.
	\end{enumerate}
	\section{سوال سوم}
	
	الکترون ها بايد تحت چه اختلاف پتانسيلی شتاب بگيرند تا بتوان از آن ها براي تفکيک و مشاهده اجسام زير با استفاده از طول موج دوبروي استفاده کرد؟
	
	\begin{enumerate}
		\item[(الف)] يک ويروس با قطر \(15\,\mathrm{nm}\)
		\item[(ب)] يک اتم با قطر \(0.096\,\mathrm{nm}\)
		\item[(ج)] يک پروتون با قطر \(1.2\,\mathrm{fm}\)
	\end{enumerate}
	
	فرض کنيد حد تفکيک مورد نياز با طول موج دوبروي الکترون ها قابل مقايسه با ابعاد ذره مورد مطالعه باشد.
\section{سوال چهارم}

سنگی به درون يک سطح آب پرتاب می شود و در محل برخورد، موجی ايجاد می کند که به مدت \(5.2\,\mathrm{s}\) ادامه دارد. سرعت انتشار موج روی سطح آب برابر \(29\,\mathrm{cm/s}\) است.

\begin{enumerate}
	\item[(الف)] گروه موج های ايجاد شده روی سطح آب چه طولی را اشغال می کنند؟
	
	\item[(ب)] يک ناظر \(12\) قله موج را در اين گروه موج مشاهده می کند. با استفاده از اصل عدم قطعيت برای بسته های موج، دقتی را که می توان برای اندازه گيری طول موج به دست آورد، تخمين بزنيد.
\end{enumerate}
	\section{سوال پنجم}
	
	ذره \(\Sigma^*\) دارای انرژی سکون \(1385\,\mathrm{MeV}\) و عمر متوسط \(2.0 \times 10^{-23}\,\mathrm{s}\) است.
	
	با توجه به اصل عدم قطعیت انرژی زمان، بازه معمول نتایج اندازه گیری انرژی سکون این ذره را تخمین بزنید.
	\section{سوال ششم}
	
	ذرات آلفا در فرایندهاي واپاشي هسته اي با انرژي هاي معمول حدود \(5\,\mathrm{MeV}\) گسيل مي شوند.بازه انرژي جنبشي يک ذره آلفا که در يک ناحيه هم اندازه هسته محبوس شده است را به دست آوريد.
	
	سپس نتيجه بگيريد که آيا يک ذره آلفا مي تواند به طور پايدار داخل هسته وجود داشته باشد يا خير.
	
\section{سوال امتیازی اول}

با در نظر گرفتن یک توزیع از عدد موج ها با دامنه ثابت در بازه ای به عرض \(\Delta k\) حول مقدار \(k_0\)، یعنی:

\[
A(k) =
\begin{cases}
	A_0 & k_0 - \dfrac{\Delta k}{2} \le k \le k_0 + \dfrac{\Delta k}{2} \\
	0 & \text{\lr{otherwise}}
\end{cases}
\]

تابع موج به صورت زیر تعریف می شود:

\[
y(x) = \int A(k)\cos(kx)\,dk
\]

با استفاده از این رابطه، تابع \(y(x)\) را به دست آورید و نشان دهید شکل موج حاصل به صورت یک بسته موج \lr{(wave packet) }خواهد بود.

	\section{سوال امتیازی دوم}
	
	با استفاده از توزیع عدد موج ها به صورت
	
	\[
	A(k) = A_0 \exp\left[-\frac{(k-k_0)^2}{2(\Delta k)^2}\right], \qquad -\infty < k < +\infty
	\]
	
	نشان دهید تابع موج به صورت زیر به دست می آید:
	
	\[
	y(x) = A_0 \Delta k \sqrt{2\pi}\; e^{-\frac{(\Delta k)^2 x^2}{2}} \cos(k_0 x)
	\]
