	\section{سوال اول}
	ثابت کنید:
	\begin{equation}
		\int_{-\infty}^{\infty} e^{-(a x^2 + b x + c)}\,dx=\sqrt{\frac{\pi}{a}}\,\exp\Big(\frac{\Delta}{4a}\Big)
	\end{equation}
	
	
	\section{سوال دوم}

		
		تابع چگالی احتمال انرژی مولکول‌ها در گاز ایده‌ال در دمای $T$:
		
		\[
		f(E) = \frac{2}{\sqrt{\pi}} \frac{1}{(k_B T)^{3/2}} \, E^{1/2} \, \exp\left(-\frac{E}{k_B T}\right), \quad E \ge 0
		\]
		
		\subsection*{الف) نرمالیزه بودن}
		\[
		\int_0^\infty f(E) \, dE = 1
		\]
		
		\subsection*{ب) ممان اول (میانگین انرژی)}
		\[
		\langle E \rangle = \int_0^\infty E \, f(E) \, dE
		\]
		
		\subsection*{پ) ممان دوم (میانگین مربع انرژی)}
		\[
		\langle E^2 \rangle = \int_0^\infty E^2 \, f(E) \, dE
		\]
		
		\subsection*{ت) واریانس انرژی}
		\[
		\sigma_E^2 = \langle E^2 \rangle - \langle E \rangle^2
		\]
		
		
		
	\section{سوال سوم}

	
	یک اتم هلیوم ($m = \SI{6.6465e-27}{kg}$) به طور کشسان با یک اتم اکسیژن ($m = \SI{2.6560e-26}{kg}$) که در حال سکون است، برخورد می‌کند. پس از برخورد، مشاهده می‌شود که اتم هلیوم با سرعت $\SI{6.636e6}{m/s}$ در جهتی با زاویه $84.7^{\circ}$ نسبت به جهت اولیه خود حرکت می‌کند. اتم اکسیژن در زاویه $-40.4^{\circ}$ حرکت می‌کند.
	\\

	\section{سوال هفتم}
	
	اتمی با جرم $m_1 = m$ در جهت مثبت $x$ با سرعت $v_1 = v$ حرکت می‌کند. این اتم با اتمی به جرم $m_2 = 2m$ که در جهت مثبت $y$ با سرعت $v_2 = 2v/3$ حرکت می‌کند، برخورد کرده و به آن می‌چسبد. سرعت حاصل و جهت حرکت ترکیب را پیدا کنید، و همچنین انرژی جنبشی تلف شده در این برخورد ناکشسان را بیابید.
